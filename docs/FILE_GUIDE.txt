🚀 QUICK COMMANDS
═══════════════════════════════════════════════════════════════════════════

# Setup & Run
  createdb dundermifflin
  go mod download
  go run main.go

# Test Login
  curl -X POST http://localhost:3000/api/v1/auth/login \
    -H "Content-Type: application/json" \
    -d '{"username":"admin","password":"Admin123!"}' \
    -c cookies.txt

# Create Product
  curl -X POST http://localhost:3000/api/v1/products \
    -H "Content-Type: application/json" -b cookies.txt \
    -d '{"sku":"PAPER-001","name":"Copy Paper","description":"20lb",
         "category":"Office","quantity":100,"low_stock_threshold":20,
         "price":8.99}'

# View Products
  curl http://localhost:3000/api/v1/products -b cookies.txt

# View Low Stock Alerts
  curl http://localhost:3000/api/v1/alerts -b cookies.txt

# View Audit Logs (admin only)
  curl http://localhost:3000/api/v1/audit-logs -b cookies.txt
